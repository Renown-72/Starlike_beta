namespace = xingxi
# 现在开始事件2，它将创建星系并把我们的星系相连接
country_event = {
    # 执行国家事件
    id = xingxi.1
    # id如1.txt
    is_triggered_only = yes
    hide_window = yes
    fire_only_once = yes
    # 如1.txt
    trigger = {
        # 条件，拥有chuangjiandekongzhi国家标志，它才会执行创建，如果没人选系外星系进入游戏
        # 那么就不会在chuangjian.1事件里给root.owner控制者设置country_flag
        # 那么大地图上就不会创建系外星系
        # 反之，删除这个trigger和它的country_flag，条件，无论有没有人选择这个系外星系进入游戏
        # 都会创建系外星系
        has_country_flag = chuangjiandekongzhi
    }
    immediate = {
        # 开始执行效果
        set_spawn_system_batch = begin
        # set_spawn_system_batch表示星系创建从此处开始,begin表示开始
        no_scope = {
            # 重置区域,将其从这个事件的执行者调整至整个大地图
            # 所以spawn_system时,this里指向的是galactic_object,也就是银河系大地图
            spawn_system = {
                # spawn_system代表开始创建星系了，第一个星系
                # 也就是放迁移后人口的那个星系,它将成为第二个象限中心点
                min_distance >= 600
                max_distance <= 600
                # 现在是第一个象限,以大地图中心为原点
                # min_distance表示这个星系距离银心的由多近,设置成0,0就是在银心上,不过不建议
                # 放在银心，会和裂隙星系起冲突
                min_orientation_angle = 120
                # min_orientation_angle代表距离银河系黑洞的最小角度
                # 90＜134＜180，所以在第二象限，即左上角
                max_orientation_angle = 120
                # 距离银河系黑洞的最大角度，同上
                initializer = zijianxingxi_1
                # initializer引入创建的那个星系
                # 也就是星系2
                hyperlane = no
                # 是否有超空间轨道？no代表没有，这里代表是否有接入大地图的超空间航道
                effect = {
                    # 结果触发
                    save_global_event_target_as = xingxi1
                    # 创建一个全局的event_target
                }
            }
            event_target:xingxi1 = {
                # 我们新建的星团使用上面创建的event_target
                spawn_system = {
                    # 再次开始创建星系3
                    min_distance >= 30
                    max_distance <= 30
                    # 这个就是距离我们创建的星团的距离了，这里以zijianxingxi_1为中心点
                    # 不是银河系中心了
                    min_orientation_angle = 90
                    # 这个就是距离我们创建的星团的角度了，不是银河系中心了
                    max_orientation_angle = 90
                    initializer = zijianxingxi_2
                    # 星系名，以下重复。
                    hyperlane = no
                    effect = {
                        save_global_event_target_as = xingxi2
                        create_starbase = {
                            # 这是给星系3创建的恒星基地

                            # 3.14可以写其他的,但4.0版本以后代码条件疑似变得更严格了
                            # 所以如果没有其它的恒星基地,如:擎天堡科技什么的,那么这里就只写哨站或者星港
                            # 这两个类型即可

                            size = starbase_outpost
                            owner = event_target:chuangjiandekongzhi
                            # 控制着，是chuangjian.1事件创建的global_event_target，
                        }
                    }
                }
                # 这里，写入星系4，和星系3一样都在 event_target:xingxi1 = {}中
                # 这样创建成的星系团会以星系1为圆点创建，方便固定星系后变得更加美观，如
                # 这样创建的星系团会始终固定在一个位置上且一直生成在那里
            }
        }
        event_target:xingxi1 = {
            # 现在开始从星系1开始创建连接星系2的超空间航道
            add_hyperlane = {
                # add_hyperlane，代表超空间航道
                from = event_target:xingxi1
                # from代表从那里开始创建
                to = event_target:xingxi2
                # to代表着开始创建的那个星系应该把超空间航道连接至哪个星系
                # 这里的event_target，是在创建星系时给星系创建的event_target
            }
        }
        set_spawn_system_batch = end
    }
}
