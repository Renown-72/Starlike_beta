# 自建护盾和A槽
# A槽和护盾都是使用的utility_component_template来编写
utility_component_template = {
    key = "ZIJIANWUQI_A"
    # 武器的KEY值，很重要，唯一值
    size = aux
    # 武器的槽位，aux代表A槽
    icon = "GFX_zijian_tu"
    # 武器的图片
    icon_frame = 1
    # 1就行
    power = -10
    # 武器锁消耗的能源，正数是产出，负数是消耗
    resources = {
        # 资源
        category = ship_components
        # 类别
        cost = {
            alloys = 4
        }
    }
    modifier = {
        # 这个武器的BUFF修正
        ship_speed_mult = 1
    }
    # class_restriction = { shipclass_military }
    # class_restriction是类别限制，
    # 一般用来限制这个部件只能放在什么舰船上
    # shipclass_military是普通的舰船，shipclass_starbase是空间站，如星港擎天堡等等
    # 一般不用它
    prerequisites = { "tech_afterburners_1" }
    # 这个部件所需要的科技
    component_set = "ZIJIANWUQI_A"
    # 注册这个部件
    # upgrades_to = "AFTERBURNER_2"
    # 这个部件升级到哪一个武器

    friendly_aura = {
        # 光环，也就是我们常说的什么什么光环了，可以直接给这个星系的友方舰船修正加成
        # 你也可以给我方负面加成，只要你觉得好玩
        #此为敌对光环，friendly_aura为友军光环
        name = "ZIJIANWUQI_A_YOUFANG"
        # 光环的名字
        radius = 300000
        # 光环的半径，很抽象的玩意，干脆使劲写0吧
        system_wide = yes
        # 系统范围，写yes就行
        apply_on = fleets
        # 适用于舰队
        stack_info = {
            id = ZIJIANWUQI_A_1
            # ID应该唯
            priority = 1
            # 1就行
        }
        modifier = {
            # 在这里写个modifier，里面就是给与友军的BUFF修正了
            ship_shield_mult = 100
        }
        graphics = {
            # 直接复制即可，抽象的图形，搞不懂蠢驴写这个是想干嘛
            area_effect = {
                entity = "circle_area_entity"
                dynamic_scale = yes
            }
            ship_effect = {
                entity = "ship_aura_negative_entity"
                dynamic_scale = no
            }
        }
    }
    hostile_aura = {
        # 光环，也就是我们常说的什么什么光环了，可以直接给这个星系的敌对舰船负面修正加成
        # 你也可以给对面正面加成，只要你觉得好玩
        #此为敌对光环，friendly_aura为友军光环
        name = "ZIJIANWUQI_A_DIFANG"
        # 光环的名字
        radius = 300000
        # 光环的半径，很抽象的玩意，干脆使劲写0吧
        system_wide = yes
        # 系统范围，写yes就行
        apply_on = fleets
        # 适用于舰队
        stack_info = {
            id = ZIJIANWUQI_A_2
            # ID应该唯
            priority = 1
            # 1就行
        }
        modifier = {
            # 在这里写个modifier，里面就是给与友军的BUFF修正了
            # 正数就是加成，负数就是减成，加成一般给友方，减一般给敌方
            ship_shield_mult = -100
        }
        damage_per_day = {
            # 光环里给敌方造成伤害的，也就是戍邦的烧血
            damage = { min = 3000 max = 3000 }
            # 造成的伤害
            accuracy = 30
            # 准确性
            shield_penetration = 30
            # 护盾穿透
            armor_penetration = 30
            # 装甲穿透
            shield_damage = 300
            # 护盾伤害
            armor_damage = 300
            # 装甲伤害
            hull_damage = 300
            # 船体伤害
        }
        graphics = {
            # 直接复制即可，抽象的图形，搞不懂蠢驴写这个是想干嘛
            area_effect = {
                entity = "circle_area_entity"
                dynamic_scale = yes
            }
            ship_effect = {
                entity = "ship_aura_negative_entity"
                dynamic_scale = no
            }
        }
    }

    ai_weight = {
        # AI权重和自动使用权重相关
        weight = 0
    }
    # 这个敌方和友方光环不仅可以放在A槽，也可以放在侧槽，护盾
    # 甚至是武器上（精神错乱）？
    # 但一般还是放在A槽和侧槽上吧
}
