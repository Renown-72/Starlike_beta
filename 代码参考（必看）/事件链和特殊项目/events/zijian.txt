namespace = zijiande
# 注册事件名，现在来引入我们的特殊项目和事件链

country_event = {
    id = zijiande.1
    hide_window = yes
    is_triggered_only = yes
    # 依次是注册事件ID，不需要弹窗和，使其事件不随便触发
    immediate = {
        # 使用enable_special_project来引入创建你得特殊项目
        enable_special_project = {
            # name写入你得特殊项目的KEY值
            name = ZIJIAN_0
            # owner是控制着，这里一般写root，也就是控制着你是你自己
            # 给你自己添加特殊项目
            owner = root
        }
    }
}
country_event = {
    id = zijiande.2
    # 第二个事件
    hide_window = yes
    is_triggered_only = yes
    # 依次是注册事件ID，不需要弹窗和，使其事件不随便触发
    immediate = {
        # 现在是事件链，使用begin_event_chain创建一个事件链
        begin_event_chain = {
            # 使用event_chain来引入你创建的事件链名字
            event_chain = become_zijiande
            target = root
        }
        add_event_chain_counter = {
            # add_event_chain_counter是给你上方begin_event_chain创建的事件链接添加一个
            # enable_special_project，也就是特殊事件，
            # 也就是把下面创建的特殊事件整到一个事件链项目里
            event_chain = "become_zijiande"
            # 使用event_chain来引入你创建的事件链名字
            counter = "wodezijian"
            # counter就是引入我们的
            amount = 1
            # amount是数量，这里就是你下面写了几个enable_special_project，这里就写几
        }
        # 使用enable_special_project来引入创建你得特殊项目
        enable_special_project = {
            # name写入你得特殊项目的KEY值
            name = ZIJIAN_1
            # owner是控制着，这里一般写root，也就是控制着你是你自己
            # 给你自己添加特殊项目
            owner = root
        }
        # 1
    }
}
