# 自建插槽和插槽数量和插槽的部件
# 插槽在common文件夹下的section_templates文件夹中编写，
# 记得新开文件夹，直接在游戏源文件复制下来的文件里改会不生效，或许改名会生效？我不想试
# 以战列舰来演示

# 一个可以建造的舰船不仅需要ship_sizes舰船文件还需要section_templates舰船插槽文件
# 这里的插槽是原来演示的代码,我只更改了fits_on_slot和ship_size
# 还有刚改的entity

ship_section_template = {
    # ship_section_template表示新建一个插槽位置
    key = "BATTLESHIP_ZIJIAN"
    # 插槽的KEY值，需要唯一值
    ship_size = zijiandejianchuan
    # ship_size，舰船类型，这里一般绑定到对应的舰船上，比如这里就是绑定战列舰battleship
    # 舰船类型在common文件夹下的ship_size文件夹中
    fits_on_slot = mid
    # 插槽所在的位置，一般bow代表左边的插槽，mid表示中间的插槽，stern表示右边的插槽，
    # 这里以右边的插槽来演示
    should_draw_components = yes
    # 是否绘制组件,复制即可
    entity = "wodezijiande"
    # 插槽的实体,没有自建实体的话复制对应位置的插槽实体即可
    # 一般只需要更改插槽中的entity,复制到里面即可
    icon = "GFX_ship_part_core_stern"
    # 插槽图片,同上
    ai_tags = { gunship carrier }
    # ai_tags,一般是给AI看的,不用管,复制即可

    component_slot = {
        # 在ship_section_template中component_slot代表这个插槽的组部件
        # 可以直接去复制原版进行修改,比如我们需要X槽,而脊峰舰艏有X槽
        # 插槽的部件都是从上到下进行排列的,如航母槽,是S,S,P,P,M,M
        # 那么第一个component_slot是S,第二个是S,第三个是P槽位,以此类推
        name = "EXTRA_LARGE_01"
        template = "invisible_extra_large_fixed"
        locatorname = "xl_gun_01"
        # 1
    }
    component_slot = {
        # 后面开始一般只需要修改数字即可,如第一个是EXTRA_LARGE_01,那么第二个是EXTRA_LARGE_02
        # 只需要修改name和locatorname
        name = "EXTRA_LARGE_02"
        template = "invisible_extra_large_fixed"
        locatorname = "xl_gun_02"
    }
    component_slot = {
        name = "EXTRA_LARGE_03"
        template = "invisible_extra_large_fixed"
        locatorname = "xl_gun_04"
    }
    component_slot = {
        name = "EXTRA_LARGE_05"
        template = "invisible_extra_large_fixed"
        locatorname = "xl_gun_05"
    }
    component_slot = {
        name = "EXTRA_LARGE_06"
        template = "invisible_extra_large_fixed"
        locatorname = "xl_gun_06"
    }
    # 现在我们创建了6个X槽,再来四个P槽位,已知航母槽有P槽部件,那么复制即可
    # 从上至下第三个是P槽位,因为没有SMALL_GUN_01,那么我们可以修改为SMALL_GUN_01
    component_slot = {
        name = "SMALL_GUN_01"
        template = "point_defence_turret"
        locatorname = "small_gun_01"
    }
    component_slot = {
        name = "SMALL_GUN_02"
        template = "point_defence_turret"
        locatorname = "small_gun_02"
    }
    component_slot = {
        name = "SMALL_GUN_03"
        template = "point_defence_turret"
        locatorname = "small_gun_03"
    }
    component_slot = {
        name = "SMALL_GUN_04"
        template = "point_defence_turret"
        locatorname = "small_gun_04"
    }
    # 四个P槽,够了

    large_utility_slots = 10
    medium_utility_slots = 5
    small_utility_slots = 1
    aux_utility_slots = 10
    # large_utility_slots的值代表这个插槽下面对应的护盾有多少,
    # large_utility_slots表示L槽护盾,
    # medium_utility_slots表示M槽护盾,
    # small_utility_slots表示S槽位护盾,它们的值代表有多少个对应的插槽
    # aux_utility_slots表示A槽位数量,现在这里有十个L护盾,五个M护盾,1个S护盾和十个A槽

    prerequisites = {}
    # prerequisites对应科技,表示这个插槽需要什么科技才能解锁

    resources = {
        # 资源
        category = ship_sections
        # 类别,ship_sections表示组件
        cost = {
            # 消耗
            alloys = @section_cost
        }
    }

    # 进游戏看看,插槽已经装上去了,为了注册插槽名字,使用BATTLESHIP_ZIJIAN值即可在localisation文件夹中注册
    # 这里就不演示了
}
