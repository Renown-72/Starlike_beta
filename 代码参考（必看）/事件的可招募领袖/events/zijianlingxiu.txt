# 事件领袖，在游戏中出现并拥有自己专属的对话框，并可以选择招募或者是不招募的领袖
# 需要events文件夹下的事件文件，common文件夹下的scripted_effects脚本文件
# common文件夹下的traits领袖特质文件，这里特质直接引用我的另一个MOD，你们需要自己写领袖特质
# 还需要gfx文件夹下的models文件夹下的portraits文件夹下的任意文件夹来存放领袖的图片
# 图片格式建议为.dds格式，大小自己测吧，这种图片规格不一定需要完全一样的
# 还需要gfx文件夹下的portraits文件夹下的portraits文件夹下的文件，
# 用来注册领袖图片
# 然后命名一个事件
namespace = zijianlingxiu

country_event = {
    id = zijianlingxiu.1
    # 注册事件ID
    title = "zijianlingxiu.1.title"
    # title是事件的标题
    desc = "zijianlingxiu.1.desc"
    # desc是事件的描述
    show_sound = par_portrait_12
    # 这个事件开启的声音，随便选一个，不使用也行
    is_triggered_only = yes
    # is_triggered_only是这个事件需要另一个事件来触发或者控制台手动触发，
    # is_triggered_only = yes也可以使其不随随便便一直触发

    picture = GFX_leader_recruitment_bg_renowned
    # 这个事件招募的背景图片，可以自己去选一个也可以自己去做一个
    auto_opens = no
    # 是否自动开启事件？这里的NO一般和notification_event_icon这个图一起使用
    notification_event_icon = "GFX_message_leader_recruitment_lysator"
    # notification_event_icon小弹窗的图片
    event_window_type = leader_recruit
    # event_window_type代表领袖事件窗口的类型，leader_recruit代表招募领袖，
    # 直接复制并使用它即可

    immediate = {
        zijianlingxiu_leader_creator = {
            # 是脚本文件，用来对应领袖招募的，这个脚本我们可以直接复制蠢驴的脚本直接使用
            # ctrl + 鼠标左键快速跳转到对应的脚本文件处
            SPECIES_CLASS = random_non_machine
            # SPECIES_CLASS是用来设置这个领袖的物种，这里我的领袖是生物，不用改直接使用它
            LEADER_PORTRAIT = NAME_MANTOUJIANONG
            # LEADER_PORTRAIT代表领袖的图片，我们刚刚创建的领袖图片可以用上了
            CLASS = commander
            # CLASS，这里的CLASS代表领袖的职责，
            # 如commander是指挥官等等，另外两个我在领袖特质制作里讲过，这里以指挥官来演示
            NAME = NAME_MANTOUJIANONG
            # NAME是领袖的名字，这里直接使用我注册的名字
            LEVEL = 4
            # LEVEL代表等级，4就是4级
            GENDER = female
            # GENDER代表领袖的性别male是男性,female是女性,indeterminable是无法确定
            # 这里使用女性
            IMMORTAL = yes
            # 这个领袖是否不朽，no代表不是,yes代表是
            ETHIC = xenophile
            # ETHIC是领袖的思潮，xenophile代表亲外主义，这里直接使用这个亲外演示
            DESC = NAME_MANTOUJIANONG_1
            # DESC是领袖的介绍
            CATCH_PHRASE = NAME_MANTOUJIANONG_2
            # CATCH_PHRASE是领袖的名人名言，这里直接复制我注册的中文
        }
        last_created_leader = {
            # last_created_leader代表给领袖加上对应的特质
            add_trait = leader_trait_mantoumonvhuashen
            add_trait = leader_trait_mantoumeihuole
            add_trait = leader_trait_mantoushenglijian
            # 这里的特质我直接引入我的另一个MOD写好的领袖特质，你需要自己写喜欢的特质
            set_age = 16
            # set_age代表给领袖的年龄，这里设置成16
            # 这样领袖就写好了，改写下面的招募和不招募和对话框了
        }
    }
    # 对话框我们需要复制两个特殊的对话框

    # 我们还需要使用对应的picture_event_data来引入我们的领袖到脚本中
    picture_event_data = {
        portrait = event_target:ethic_leader
        # portrait下的event_target:ethic_leader需要写你复制的那个事件的值
        room = ethic_spaceship_room
        # room的图片可以随便选择
    }

    option = {
        name = DISMISS
        # DISMISS是这个对话框的忽略选项
        tag = dismiss_leader
        # 对话框的tag，可以不用动他
        hidden_effect = {
            remove_global_flag = zijianlingxiu_leader_1 # Put this leader back into the pool, so player could still get
            # him
            # 这里建议修改一下remove_global_flag，set_country_flag
            # 和下面的GLOBAL_EVENT_TARGET = renowned_xenophile1的值，不要与原版复制的起冲突，
            # 三个值写与你领袖相关的即可，方便找到，
            # 更改它们的值后就不用关了
            zijianlingxiu_leader_kill_effect = yes
            # 我们可以看到这里还有两个脚本，也就是我们还需要复制两个脚本
            # 将更改的脚本放到对应的位置，还有一个
            set_country_flag = zijianlingxiu_xenophile_leader_1
        }
    }
    # 然后是领袖下面的对话，这里在两个option中间写吧，我看蠢驴也是这么写，它这么做应该有它的大病
    # 对话也是使用option即可
    # 这里写三个对话来演示
    option = {
        name = zijianlingxiu.1.a
        # name是你得对话的选项
        response_text = zijianlingxiu.1.a.a
        # response_text是你点击这个对话后它上方回答你选择的选项的对话
        is_dialog_only = yes
        # is_dialog_only的意思是仅用于对话，直接使用它即可，我们去复制写好的对话
    }
    option = {
        name = zijianlingxiu.1.b
        response_text = zijianlingxiu.1.b.a
        is_dialog_only = yes
    }
    option = {
        name = zijianlingxiu.1.c
        response_text = zijianlingxiu.1.c.a
        is_dialog_only = yes
    }
    option = {
        name = HIRE
        # HIRE是这个对话框的招募选项
        tag = hire_leader
        # 对话框的tag，可以不用动他
        custom_tooltip = NAME_MANTOUJIANONG_JIARU
        # 这里的custom_tooltip的意思是在你得鼠标移动向选项选项会出现一个文本，
        # 一般是用来介绍事件或者是介绍这个按钮的作用是什么
        # 这里直接使用注册好的文本
        hidden_effect = {
            zijianlingxiu_leader_hire_effect = {
                # 复制到对应位置即可
                GLOBAL_EVENT_TARGET = zijianlingxiu_xenophile1
            }
        }
    }
    # 这样我们的领袖就写好了，进游戏看看，忘了设置一下
}
