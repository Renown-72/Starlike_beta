# 事件1
planet_event = {
    id = zijiande.1
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        # 先以事件1为例,你可以看到这个插件已经给你标出来这个事件的作用域了
        # 所以你在下面像这样写的话,你会发现它们因为和planet相等于,所以也是能直接代表planet的
        # -------------------------------------- #
        # 但现在我们如果想要在这个planet_event里检测我们的国家,也就是country的条件怎么办呢?
        # 这时候我们就需要进行【跨域】了,也就是 跨作用域,已知this是这个行星,那么我们可以使用
        # this.owner,这个东西的意思是什么呢?
        # this,代表这个行星, - . ,代表的 - owner,代表控制者
        # 合起来就是这个行星的控制者,这算是很经典的跨域了,当然控制者不止owner一种,还有如下几个
        # branch_office_owner,planet_owner和space_owner
        # 拿去翻译可知branch_office_owner是分公司所有者,也就是分部,那我们肯定用不到它
        # planet_owner是星球所有者,我们可以用到它
        # space_owner是空间所有者,也就是星系的所有者
        # 因为我们的星系可能有各种各样的因素导致所有者不正确,所以我们这里不使用它
        # 所以我们可以使用owner或是planet_owner,那这里为了明显使用planet_owner吧

        # 开局的检测好像有控制权问题,所以这里先使用owner

        this.owner = {
            # 这里我们使用协同进化国家作用域的【被检测条件】
            # 也就是被这个事件检测的条件来进行检测(因为它有第二物种)
            has_origin = origin_syncretic_evolution
        }


        # -------------------------------------- #
        from = {
            is_species_class = HUM
        }
        fromfrom = {
            is_species_class = HUM
        }
        # -------------------------------------- #
        # 这里说一下,一般基础的事件不做任何"处理"的话,你这个事件是没有像这样的fromfrom什么的
        # 只有基础的作用域,因为这里的事件是由接口empire_init_capital_planet来触发的
        # 所以它带有【from】和【fromfrom】,分别代表【主体物种】和【次级物种】
        # -------------------------------------- #
    }
    immediate = {
        # -------------------------------------- #
        # 而我们的【被检测条件】和【被执行的代码】也是有作用域的
        # 这里先以【被执行的代码】为例子(被执行的代码也就是等下会被这个事件执行的代码,如下)
        # change_planet_size就是要被这个事件执行的代码,也就是【被执行的代码】
        # 现在把鼠标移动到这个代码上我们可以看到
        # 和上面的这个作用域一样,所以我们可以在这个【直接】执行这个【被执行的代码】
        # -------------------------------------- #
        # 现在让我们进入一下on_actions,也就是接口文件夹,等下我们需要让接口触发这个事件
        # -------------------------------------- #
        change_planet_size = 35
        # 将目标行星的尺寸增加或减少指定数值
        # set_planet_size = 35
        # # 将行星大小设置为指定数值,这里用错代码了
        # 但不要紧这里生效了,因为一般原版哪有开局给你42槽位星球的
        # -------------------------------------- #
        from = {
            save_event_target_as = zijiande_from
            # 在执行时候我们给这个from设置一个event_target
        }
        # -------------------------------------- #
        # 如果我们想给国家执行country事件那也是一样的,先[转换作用域]再[执行]
        this.planet_owner = {
            country_event = {
                id = zijiande.2
                # 事件里除了 days 和 random 和 years 等等的东西以外,还有一个名为
                # scopes的东西,我们需要使用它来基础作用域
                scopes = {
                    from = event_target:zijiande_from
                    # 这样我们的作用域就继承完毕了
                    # 事件zijiande.2里的from等于事件zijiande.1里的from
                    # 如果你看不懂的话可以这样

                    # 再往这里填入它,就能让from和from相等于了
                }
            }
        }
        # -------------------------------------- #
    }
}
# 事件2
country_event = {
    id = zijiande.2
    hide_window = yes
    is_triggered_only = yes
    immediate = {
        # -------------------------------------- #
        # 这里我们说一下作用域继承!如果我们想让与zijiande.1事件毫不相关的zijiande.2的from
        # 也代表这个帝国的主体物种,那么我们该怎么办呢?这时候就需要作用域继承了
        # -------------------------------------- #
        # 因为事件再执行后会完美的基础作用域,所以我们在这里直接进行执行条件
        # 注意一下阿, 因为 from = species,所以这里继承过来的from是这样的
        # from = { this = species }
        from = {
            # 换一个,我不太会使用modify_species
            change_species_characteristics = {
                add_trait = trait_very_strong
                add_traits_at_start_of_list = yes
            }
        }
        # -------------------------------------- #
        # 现在继续转换作用域以此来执行星系事件
        # random_owned_planet = {
        #     # 我拥有的其中一个行星
        #     limit = {
        #         # 条件,需要为首都
        #         is_capital = yes
        #     }
        #     # 这个行星 的 星系
        #     # 注意,这里的solar_system代表的是星系,而不是翻译的太阳系
        #     this.solar_system = {
        #         system_event = {
        #             id = zijiande.3
        #         }
        #     }
        # }
        # -------------------------------------- #
        # 你如果会用翻译的话你也可以这么写
        # capital_scope翻译为资本范围,也就是首都的范围
        # capital_star翻译为首都星,应该是首都的恒星,这里我用它用的少,所以使用capital_scope
        this.capital_scope.solar_system = {
            # 翻译可知 这个帝国 的 首都 的 星系
            system_event = {
                id = zijiande.3
            }
        }
        # -------------------------------------- #
    }
}

        }
    }
}
