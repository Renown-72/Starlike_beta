# 事件文件
namespace = zijiande
# 用了左边这些文件
# 用接口来检索拥有的物种来创建对应的人口
planet_event = {
    id = zijiande.1
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        this.owner = {
            has_origin = origin_zijian
        }
    }
    immediate = {
        this.owner = {
            # 将作用域从行星转为国家
            every_owned_species = {
                # 我的所有物种
                limit = {
                    # 进行条件检索
                    # 有着奴性特质才是我要找的
                    has_trait = trait_syncretic_proles
                }
                save_event_target_as = this_is_my_secondary_species
                # 设置一个event_target方便等下创建的时候找到这个物种
            }
        }
        # 请注意4.0以前是create_pop,4.0后是create_pop_group,事件代码以当前的版本为准
        # 代码的样子一般不会偏离太多
        create_pop_group = {
            # 用create_pop_group来创建人口
            species = event_target:this_is_my_secondary_species
            # 创建的人口是什么物种?用event_target快速找到它!
            # 注意这里的event_target不是全局的,一般只能在这个事件里使用
            size = 1500
        }
        # 这样创建人口就写好了
    }
}
# 用控制台来创建物种并创建对应的人口
# 如果是新建一个新的物种呢?
country_event = {
    id = zijiande.2
    hide_window = yes
    is_triggered_only = yes
    immediate = {
        # 那就得先创建物种再来创建人口了
        # 以原版的这个为例
        create_species = {
            name = NAME_Cetanas_Prisoners
            # 这里是创建的物种的名字
            class = "MACHINE"
            # class里写了什么类型的物种等下你创建的人口就是什么类型
            # 这里很明显是机械
            portrait = "synth_machine_02"
            # 物种的图像,如果你不知道你得图像你可以创建一个预设国然后看它的portrait值是什么
            traits = {
                # 创建的物种有哪些特质
                trait = trait_mechanical
                trait = trait_cetanas_prisoner
            }
            homeworld = this.capital_scope
            # 它的母星是哪里,capital_scope的意思是这个帝国的母星,因为等下是我执行的事件
            # 所以意思就是我的母星是它的母星
            effect = {
                # 同理需要一个event_target
                save_event_target_as = this_is_my_secondary_species_2
            }
        }
        # 因为这里不是行星作用域,所以要用代码找到对应的行星
        every_owned_planet = {
            # 我所有的行星,请注意,虽然是所有的行星,但根据注释我们可以知道
            limit = {
                # 只找母星,只给母星创建人口
                is_capital = yes
            }
            create_pop_group = {
                size = 1500
                species = event_target:this_is_my_secondary_species_2
            }
        }
    }
}
