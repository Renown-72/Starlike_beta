# 星球地块，在common文件夹下的deposits文件中编写
d_zijian = {
    # 代表这个是什么样式的地块，如资源地块，普通地块，和这次使用的可铲除地块等等
    category = deposit_cat_blockers_natural
    # 和上方的deposit_cat_blockers_natural一起使用，代表这个地块铲除所需的时间
    time = 300
    # 地块的图片，这里直接使用原版
    icon = "d_station_reactor"
    # 地块生成所需的条件，例如
    potential = {
        planet = {
            # 星球格子大于等于十才会生成
            planet_size >= 10
        }
    }
    # 铲除所需的条件，和deposit_cat_blockers_natural地块配合使用，例如
    can_be_cleared = {
        # 意思是，planet，在这个行星上
        planet = {
            # NOT，没有
            NOT = {
                # 合金铸造厂，即可铲除
                has_building = building_foundry_1
            }
        }
    }
    # 铲除后发生的效果，
    on_cleared = {
        # 在这个行星上
        planet = {
            # 添加一个合金铸造厂
            add_building = building_foundry_1
        }
    }
    # 行星能吃到的效果修正，一般不配合deposit_cat_blockers_natural可清除地块使用
    # 因为地块铲除后效果也会消失，这里只是演示
    planet_modifier = {
        # 住房增加
        planet_housing_mult = 100
    }
    # 条件通过后能够获得的效果，例如，同上
    # 一般不配合deposit_cat_blockers_natural可清除地块使用
    triggered_planet_modifier = {
        potential = {
            planet = {
                # 这个行星上有个合金铸造厂
                has_building = building_foundry_1
            }
        }
        # 住房再加100
        planet_housing_mult = 100
    }
    # 资源
    resources = {
        # 类别，一般配合地块的category使用，不写也行
        category = deposit_blockers_natural
        # 铲除地块的消耗
        cost = {
            # 铲除需要66能量币
            energy = 66
        }
    }
    prerequisites = {
        # 一般放置铲除地块所需的科技，这里就不写了，配合deposit_cat_blockers_natural使用
    }
}
# 地块已经在星球上了，铲除看看,正确执行了，就这样
