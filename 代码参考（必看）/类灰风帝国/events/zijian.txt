# 然后是事件,这里事件分为三个
# 第一个
# 使用事件创建可以通讯的帝国
# 首先是第一个,没什么需要太注意的
country_event = {
    id = zijianlingxiu.1
    hide_window = yes
    is_triggered_only = yes
    immediate = {
        set_country_flag = ruled_by_zijian
        # 首先给你自己创建一个flag,用于区分其它帝国
        create_country = {
            # 然后进行创建国家
            name = "zijianguojia"
            # 这里是创建的可以通讯帝国的国家名字
            type = zijiandiguo
            # 这里是国家type
            flag = {
                # 帝国的旗帜,可以自定义,但等一下再讲这个
                # 最后说一下旗帜
                # 你可以先自定义一个预设国
                icon = {
                    # 图片
                    category = "pirate"
                    file = "flag_pirate_1.dds"
                }
                background = {
                    # 背景
                    category = "backgrounds"
                    file = "00_solid.dds"
                }
                colors = {
                    # 颜色
                    "light_pink"
                    "dark_purple"
                    "black"
                    "null"
                }
            }
        }
        last_created_country = {
            # last_created_country,给创建帝国的效果
            establish_communications_no_message = root
            # 让这个创建的可通讯帝国与你开启通讯,然后你就可以
            # 直接与他通讯并开启事件进行召唤领袖了
            create_leader = {
                # 这里是给它整个领袖,就是scripted_effects下的指挥官领袖
                # 作用是什么？拿来看的,你可以对着你scripted_effects下的领袖写上就行
                class = commander
                species = root
                name = zijianlingxiu
                custom_catch_phrase = zijianlingxiudejieshao
                event_leader = yes
                skill = 10
                immortal = yes
                hide_age = yes
                gender = female
                traits = {
                    trait = leader_trait_zijian
                }
                effect = {
                    # 设置领袖的图像
                    change_leader_portrait = zijian
                }
            }
            set_leader = last_created_leader
            # 给这个帝国设置领袖
            save_global_event_target_as = magic_dp_madoka_country
            # 给这个帝国设置全局_event_target,暂时没用到它,或许以后有用
        }
    }
}

# 第二个
# 可以在通讯页面和帝国通讯开启领袖通讯事件
# 然后是第二个
country_event = {
    id = zijianlingxiu.2
    title = zijianlingxiu.2.title
    desc = zijianlingxiu.2.desc
    diplomatic = yes
    # 注意这里需要使用diplomatic来将普通的事件变为通讯事件
    picture = GFX_evt_mysterious_signal
    picture_event_data = {
        portrait = from.leader
        room = root
    }
    # 这里两个是通讯事件的图片相关,不用管
    is_triggered_only = yes
    trigger = {
        # 注意这里了,第一个检测的条件,
        has_country_flag = ruled_by_zijian
        # 我们需要这个flag才能和这个国家开启通讯事件
        from = { is_country_type = zijiandiguo }
        # 然后是这个from,
        # 检索的条件内容是检测到与这个特定的国家进行外交时候才会开启这个事件
        # 这里的country_type和上面的一样
        NOT = { has_active_event = { zijianlingxiu.2 } }
        # has_active_event
        # 然后说一下这个,这个东西是检测有无这个zijianlingxiu.2事件
        # 在NOT内的意思需要没用这个事件,总体含义就是如果有了这个zijianlingxiu.2事件
        # 你再点击外交时就不会再触发这个事件,防止你一直点外交按钮出现一堆这个事件
    }
    option = {
        # 然后是这两个切换领袖职能的按钮
        # 把它写在我们的两个切换职能的领袖按钮的trigger条件里即可
        # 写在NOR里,代表这个按钮在我们没有这个BUFF修正的时候才能使用
        # 又因为领袖死亡时会给你添加一个BUFF修正，
        # 那么也就代表我们有这个BUFF修正时候不能使用这个按钮
        name = zijianlingxiu.2.b
        response_text = zijianlingxiu.2.b.a
        trigger = {
            # trigger,注意这里的trigger的作用
            # NOR里检测我们所有领袖是否具有zijianlingxiu_zhihuiguan
            # 这个创建的领袖的专属flag，如果有，那么代表这个领袖已经被
            # 召唤出去了,那么这个按钮就不应该再出现了
            NOR = {
                has_modifier = mod_magic_gray_goo
                any_owned_leader = {
                    has_leader_flag = zijianlingxiu_zhihuiguan
                }
            }
        }
        hidden_effect = {
            # 你可以看到这里我创建了一个country_flag,但在第三行的时候又给自己删除了
            # 它的作用是什么呢？
            set_country_flag = magic_by_zijian
            # 那么就很明确了,先设置flag防止切换领袖职能的时候被检测到联系死亡了
            # 然后创建领袖
            # 最后再取消掉这个flag，防止领袖真正死亡的时候又因为这个flag检测不到领袖死亡了
            zijianlingxiu_1 = yes
            # 第一个是给指挥官的
            remove_country_flag = magic_by_zijian
        }
    }
    option = {
        name = zijianlingxiu.2.c
        response_text = zijianlingxiu.2.c.a
        trigger = {
            # 这里同理，那么这样我们的领袖的切换职能就写完了
            NOR = {
                has_modifier = mod_magic_gray_goo
                any_owned_leader = {
                    has_leader_flag = zijianlingxiu_keyanzhuanjia
                }
            }
        }
        hidden_effect = {
            set_country_flag = magic_by_zijian
            zijianlingxiu_2 = yes
            # 第二个是给行政官的
            remove_country_flag = magic_by_zijian
        }
    }
    option = {
        # 默认需要写的按钮选项,里面什么效果都不写,用来关闭事件的
        name = zijianlingxiu.2.a
        # default_hide_option = yes的意思是让这个事件的关闭按钮绑定这个选项
        # 这里说一下每个事件的右上角不是有个叉号吗,如果你点击它的话
        # 它会执行你代码里的第一个option,但如果你在代码里写了default_hide_option = yes
        # 那么无论如何你点击叉号执行的就是这个option，注意写一个就行
        # 即使你把它放在最下面点击叉号也是执行这个option
        default_hide_option = yes
    }
}
########################################################
# 第三个
# 我们的通讯帝国召唤出来的领袖的死亡事件
# 然后是第三个事件,在联系死亡的时候会弹出一个事件
# 告诉我们领袖寄了
# 得需要多久才能再次召唤
country_event = {
    id = zijianlingxiu.3
    title = zijianlingxiu.3.title
    desc = zijianlingxiu.3.desc
    picture = GFX_evt_magic_huigui
    is_triggered_only = yes
    trigger = {
        # 这里的country_flag和第二个一样，都是区别其它帝国让其它帝国不能触发这个事件的
        has_country_flag = ruled_by_zijian
        # 是给这个事件看的,我们知道我们的切换领袖是要杀死这个领袖
        # 但我们切换领袖的职能的时候不能让它(这个事件)检测啊
        # 我们就先创建一个flag等领袖创建完毕了再取消掉
        # 那么这里就不会再我们切换领袖职能导致杀死领袖的时候检测领袖了
        NOR = {
            has_country_flag = magic_by_zijian
        }
        # 前两个我已经说过了,是用来区分帝国和不让这个事件
        # 在切换领袖职能的时候触发这个事件
        from = {
            # from,是检测领袖的,这里我们不能让随便一个领袖似了就触发啊
            # 所以我们检测领袖的特殊之处,你可以检测这个通用flag或者专属flag
            # 但我这里使用专属的特质来进行检测
            # 因为这个特质是它独有的,所以拥有这个特质的领袖死亡了也只能是这个领袖死亡了
            OR = {
                has_trait = leader_trait_zijian
            }
        }
    }
    immediate = {
        # 这里说一下在领袖死亡后需要一段时间才能召唤是怎么做的
        # 就是给你自己一个modifier,BUFF修正
        # 这里我用另一个MOD的，你可以自己在static_modifiers文件夹里写一个
        # days是持续的天数,我们写个30天
        # 那么效果就很明确了,领袖死亡时候会给我们添加一个效果修正
        # 在效果修正持续的时间里我们无法召唤领袖,
        add_modifier = {
            modifier = mod_magic_gray_goo
            days = 30
        }
    }
    option = {
        name = zijianlingxiu.3.a
    }
}
