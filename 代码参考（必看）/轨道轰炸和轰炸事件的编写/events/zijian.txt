# 需要在events里编写对应的事件
# 注意，轰炸代码只需要注册图片并在bombardment_stances下编写即可，这里是相关的事件
# 不需要它轰炸也可以使用
namespace = zijianhongzha
# 注册事件名
planet_event = {
    # 因为是行星轰炸所以使用planet_event，行星事件
    id = zijianhongzha.1
    hide_window = yes
    is_triggered_only = yes
    # 分别是注册事件ID，隐藏窗口和使事件不随便触发
    trigger = {
        # 使用has_orbital_bombardment来检测这个行星上是否受到了轨道轰炸
        has_orbital_bombardment = yes
        # 使用has_orbital_bombardment_stance来检测这个行星受到了哪一种轨道轰炸
        # 后面的值引入你得轰炸名
        has_orbital_bombardment_stance = zijianhongzha
        # 使用planet_devastation来检测这个行星受到了多少轰炸
        planet_devastation >= 50
    }
    immediate = {
        # 上面的条件代码都满足后将对这个行星立即执行事件
        # 执行事件，使用change_pc来更改被轰炸星球的星球类型，从原星球变为pc_shrouded，虚境星球
        change_pc = pc_shrouded
    }
}
