# 但4.0的烧人还要有game_rules和events的支持

# game_rules需要覆盖can_pops_grow_on_planet和can_pops_assemble_on_planet
# 让星球上的人口停止增长

# This = planet
can_pops_grow_on_planet = {
    NOR = {
        # 这里用另一个MOD的建筑,来模拟这个星球有这个建筑的时候(满足了某种条件)
        # 人口停止增长
        has_building = building_magic_master_archive
        has_modifier = planet_population_control
        has_modifier = planet_population_control_gestalt
        is_planet_class = pc_cosmogenesis_world
        # 突触是直接判断星球类型
        is_planet_class = pc_nanotech
        AND = {
            # conquered by crisis
            exists = controller
            controller = {
                OR = {
                    is_country_type = ai_empire
                    is_country_type = swarm
                }
            }
        }
    }
}
# This = planet
can_pops_assemble_on_planet = {
    NOR = {
        # 这里用另一个MOD的建筑,来模拟这个星球有这个建筑的时候(满足了某种条件)
        # 人口停止增长
        has_building = building_magic_master_archive
        has_modifier = planet_robot_assembly_control
        has_modifier = planet_population_control_gestalt
        is_planet_class = pc_cosmogenesis_world
        AND = {
            # conquered by crisis
            exists = controller
            controller = {
                OR = {
                    is_country_type = ai_empire
                    is_country_type = swarm
                }
            }
        }
    }
}
