# 初步认识一下事件，事件在events文件夹中编写
namespace = zijian
# namespace，注册事件名字，是整个事件最重要的东西


country_event = {
    # country_event,执行一个国家事件
    id = zijian.1
    # 注册这个事件的id，需要事件名 + 数字 ，最好不要从零开始且数字不宜太大
    # hide_window = yes
    # 让这个事件不弹出弹窗，一般绑定yes使用，因为你如果想弹事件就得写
    # title,desc,picture,他们分别是事件的标题和事件的内容和事件的图片
    # 所以弹窗事件我们也不需要hide_window
    # picture中的图片需要去interface下注册
    # show_sound是这个事件执行后触发的音效，不写也行
    title = zijian.1.title
    desc = zijian.1.desc
    picture = GFX_evt_mining_station
    show_sound = event_ship_bridge
    is_triggered_only = yes
    # 注意这里要加is_triggered_only = yes,不然事件会一直触发,而不是按我们既定的条件和接口触发
    trigger = {
        # 因为是国家事件,我们可以直接写国家的判断条件,
        # 如:起源,国策,政体等等
        # 如果想让单独的某个起源在进入游戏执行这个事件该怎么做呢?
        # has_ + 文件的名字,一般是这样使用代码来检测
        # 而has_origin是检测你是否具有某种起源
        # 这里是让繁荣一统起源执行这个事件
        # 想让繁荣一统还是某种政体就再加AND和你得条件
        # 政体是再authorities文件夹中,所以是has_authority
        # 以此类推
        # AND是且,你需要具有某种条件且还需要具有某种条件即可
        # OR是或者,你具有某种条件或者另一种条件即可
        # NOT是不,你需要不具有某种条件才可以执行
        # NOT里也可以写OR和AND
        has_origin = origin_default
        # 这里可以不写AND直接写也行
        AND = {
            is_ai = no
        }
    }
    immediate = {
        # immediate里一般写执行的效果,也可以把效果写在option里
        # 他们的区别是immediate的效果会在这个事件执行时立刻获得,而option需要选择选项才可获得
    }
    option = {
        name = zijian.1.a
    }

    # 1
}
# 如果我们还想让我们的母星在进入游戏时候,进行某种改动怎么办?
# 因为是初始化大地图的母星,我们需要使用事件来做
# 又因为原版本身就具有初始化母星事件,所以我们直接写在母星上的大多数东西会被直接顶掉,所以我们加在母星上的东西
# 同样需要使用初始化母星事件接口才可以正常获得
# 又又因为是星球,所以我们需要一个星球事件,也就是planet_event
# 同样写事件名字和其他的,因为我们是星球事件,这里就不写事件弹窗了
planet_event = {
    id = zijian.2
    # 注意这里事件ID不能重复
    hide_window = yes
    is_triggered_only = yes
    # 注意这里要加is_triggered_only = yes,不然事件会一直触发,而不是按我们既定的条件和接口触发
    trigger = {
        # 因为是星球,我们不想直接检索星球,而检索控制者这个帝国的国家条件该怎么办呢?
        # 使用owner,这样我们在owner写入的国家条件,那么这个事件就能正确的检索这些条件了
        # 这里使用上面一样的条件,拥有某种起源且不是AI帝国
        owner = {
            has_origin = origin_default
            AND = {
                is_ai = no
            }
        }
    }
    immediate = {
        # 满足这些条件后为这个行星执行效果
        # 将行星更改为盖亚星球
        change_pc = pc_gaia
        # 将行星的大小更改为40,这里的40其实就是格子数量,星球的区划数量
        change_planet_size = 40
    }
}
# 依靠这种方法,你可以为自己执行各种各样的效果,如给你的星球加特殊建筑,为你的国家增加特殊事件和效果等等
