# 稀有文物的按钮
# 在common/artifact_actions文件夹下编写，编写方式和星球决议，法令差不多
ac_zijian = {
    resources = {
        # resources,资源消耗相关
        category = artifact_actions
        # category,类别,复制这个即可
        cost = {
            energy = 1
            # 这里的意思就是启动这个按钮需要1能量币消耗
        }
        produces = {}
        upkeep = {}
        # 从上至下依次为消耗，产出和维护,但后两个一般不用
        # 资源代码在strategic_resources下
    }
    potential = {
        # has_country_flag =
        # has_origin =
        is_ai = no
        # 例如它们,依次为检测国家的flag和需要某种起源和不是AI才可使用
    }
    # potential,这个按钮的条件
    allow = {

    }
    # allow,也是这个按钮的条件,这两个条件内都是直接作用于国家的
    ai_weight = {}
    effect = {
        custom_tooltip = ac_zijian_1
        # custom_tooltip写在这里可以用来介绍你执行了什么效果
        # effect,执行的效果
        hidden_effect = {
            # hidden_effect写了后不会显示你这个代码执行了什么效果,可写可不写
            random_system = {
                limit = {
                    is_capital_system = yes
                    # 检测是否为首都星系
                }
                solar_system = {
                    # 这个代码的意思为找到你都首都星系,
                    # 并在你这个星系中创建一个巨型船坞
                    spawn_megastructure = {
                        type = "mega_shipyard_3"
                        planet = this.star
                        # 放在这个星系的恒星上
                        owner = root
                        # 控制权为这个星系的所有者帝国
                    }
                }
            }
        }
    }
}
