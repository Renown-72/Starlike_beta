            }

        }

        event_target:sl_copan_bige_country = {
            create_species = {
                name = "sl_copan_bige_specie"
                class = HUM
                portrait = sl_copan_bige
                namelist = "MS_MAID_1"
                traits = { trait = trait_adaptive trait = trait_enduring trait = trait_weak trait = trait_hive_mind }
                effect = { save_event_target_as = sl_copan_species }
            }
            create_leader = {
                class = scientist
                name = "sl_copan_bige_name"
                tier = leader_tier_legendary
                species = event_target:sl_copan_species
                skill = 7
                gender = female
                #event_leader = yes
                skip_background_generation = yes
                background_ethic = ethic_spiritualist
                hide_age = yes
                immortal = yes
                randomize_traits = no
                traits = {
                    trait = trait_scientist_stellarite_touched
                    trait = subclass_scientist_scholar
                    trait = leader_trait_expertise_biology
                    trait = leader_trait_narrow_minded
                }
                effect = {
                    add_trait = { consume_selection = yes show_message = no trait = subclass_scientist_scholar }
                    add_trait = { consume_selection = yes show_message = no trait = leader_trait_expertise_biology }
                    add_trait = { consume_selection = yes show_message = no trait = leader_trait_narrow_minded }
                    set_leader_flag = sl_bige
                    set_leader_flag = sl_bige_speaker
                    set_leader_flag = is_in_recruit_window

                    set_leader_flag = immune_to_negative_traits

                    save_global_event_target_as = sl_copan_bige
                    set_owner = Root
                    exile_leader_as = sl_copan_bige
                }
            }
        }
    }

    event_window_type = leader_recruit
    picture = GFX_leader_recruitment_bg_ms_maid
    picture_event_data = { portrait = event_target:sl_copan_bige room = pre_ftl_modern_room }

    option = {
        name = DISMISS
        tag = dismiss_leader
        custom_tooltip = sl_bige_catch.dismiss.tt
        default_hide_option = yes
        hidden_effect = {
            event_target:sl_copan_bige = { kill_leader = { show_notification = no } }
        }
    }

    option = {
        name = HIRE
        tag = hire_leader
        custom_tooltip = sl_bige_catch.hire.tt
        hidden_effect = {
            clone_leader = { target = event_target:sl_copan_bige skip_background_generation = yes }
            last_created_leader = {
                copy_flags_and_variables_from = event_target:sl_copan_bige
                remove_leader_flag = is_in_recruit_window
                save_global_event_target_as = sl_copan_bige
            }
            create_message = {
                type = MESSAGE_RECRUITED_LEADER
                localization = MESSAGE_RECRUITED_LEADER_DESC
                days = 30
                target = event_target:sl_copan_bige
                variable = { type = name localization = LEADER scope = event_target:sl_copan_bige }
                variable = { key = "border" value = "GFX_border_legendary" }
            }
        }
    }

    option = {
        name = sl_bige_catch.intro
        is_dialog_only = yes
        response_text = sl_bige_catch.intro.r
    }

    option = {
        name = sl_bige_catch.join
        is_dialog_only = yes
        response_text = sl_bige_catch.join.r
    }

    option = {
        name = sl_bige_catch.criticize
        is_dialog_only = yes
        trigger = { OR = { is_xenophile = yes is_pacifist = yes } }
        response_text = sl_bige_catch.criticize.r
    }

    option = {
        name = sl_bige_catch.praise
        is_dialog_only = yes
        trigger = { OR = { is_xenophobe = yes is_militarist = yes } }
        response_text = sl_bige_catch.praise.r
    }

    after = { }
}

# sl_bige_cmt.10 — 调查星球锐评
country_event = {
    id = sl_bige_cmt.10
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        any_owned_leader = { is_same_value = event_target:sl_copan_bige }
    }

    immediate = {
        event_target:sl_copan_bige = {
            reroll_random = yes
            random_list = {
                6 = { }
                1 = { sl_change_bige_opi = { add = 1 } }
                1 = { sl_change_bige_opi = { add = 2 } }
            }
        }
    }
}
