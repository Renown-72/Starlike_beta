# 安全推进地球遗产事件链
sl_progress_earth_chain = {
    if = {
        limit = { has_event_chain = sl_chain_earth_legacy }
        add_event_chain_counter = {
            event_chain = sl_chain_earth_legacy
            counter = sl_earth_legacy_progress
            amount = 1
        }
    }
}
