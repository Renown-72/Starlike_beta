astral_rift_event = {
    id = zijiande.1
    title = zijiande.1.title
    desc = zijiande.1.desc
    # 这玩意你应该知道是什么意思
    # ---------------------------- #
    difficulty = 1
    # difficulty,难度,一般值是1到10
    astral_rift = yes
    # 表明你这个事件是裂隙事件
    # 虽然我不明白蠢驴都给你整了个专门的事件了还要写这个干什么
    # 总之你写上去就是了
    # ---------------------------- #
    picture = GFX_evt_astral_rift_riftworld
    show_sound = ap_rift_event_0_astral_wonder
    is_triggered_only = yes
    # 这玩意你应该知道是什么意思
    # ---------------------------- #
    option = {
        # 裂隙的选项一
        name = zijiande.1.a
        set_next_astral_rift_event = {
            # 来让你选项下个该执行的是什么
            # 你这里可以选项一执行zijiande.2,选项二执行zijiande.3
            id = zijiande.2
        }
    }
    option = {
        # 裂隙的选项二
        name = zijiande.1.b
        set_next_astral_rift_event = {
            id = zijiande.2
        }
    }
    # ---------------------------- #
    after = {
        # 这个是原版的,让你探索完一个事件后随机给你裂隙资源的
        explorer = {
            astral_thread_small_reward = yes
        }
    }
    # ---------------------------- #
}

# 事件6是拿来创建裂隙的
country_event = {
    id = zijiande.6
    hide_window = yes
    is_triggered_only = yes
    immediate = {
        # this,这个执行着|.,的|capital_scope,首都|.,的|solar_system,星系
        this.capital_scope.solar_system = {
            # 生成一个裂隙,由于使用了ID所以它是指定的裂隙,不写就是随机生成
            spawn_astral_rift = {
                id = zijiande
                random_pos = yes
                orbit_angle = 360
                spawn_sound = no
            }
        }
    }
}
