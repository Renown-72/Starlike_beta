    key = "ZIJIANWUQI_FANYINGDUI"

    # 1就行

        cost = {
            alloys = @cruiser_cost_6
            sr_dark_matter = 4

            energy = 1.48
            alloys = 0.148
        }

    # 需要的科技

        weight = 100000
    }
}
