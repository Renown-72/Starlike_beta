namespace = zijiande

country_event = {
    id = zijiande.1
    title = zijiande.1
    desc = zijiande.1
    is_triggered_only = yes
    option = {
        name = zijiande.1
    }
    option = {
        name = zijiande.2
        allow = {
            has_country_resource = {
                type = unity
                amount = value:zijiande_jugou_jiance|ZIJIANDE_JUGOU_FACTOR|5000|
                # 这里进行检测,检测你得国家资源储备,检测的资源为,凝聚力,
                # 检测的基础数量为5000,每有一个星环增加5000

                # 因为没有星环基地,所以添加了0个星环基地,并加了0次的5000凝聚力
                # 所以检测的凝聚力为0,这里一般检测为>=,我少写了,
                # 星环基地建完会摧毁巨构所以检测不到,换一个其它的检测
                # 检测大于等于5人口的星球

                # 大于等于5人口的星球是3个,所以是3个5000

            }
        }
        while = {
            # 进行一次执行,执行此处为拥有几个星环基地
            # 因为zijiande_jugou_jiance检测的是每有一个星环基地进行一次+1
            add_resource = {
                unity = -500
                mult = value:zijiande_jugou_jiance
            }
            # 减去三个500凝聚力
        }
    }
}
