zijiande_jugou_jiance = {
    # 再以此为例
    complex_trigger_modifier = {
        trigger = count_owned_planet
        # 进行检测,检测的条件为我们拥有的星球,count_owned_planet也一般代指我们拥有的殖民地
        trigger_scope = owner
        # 检测的范围,殖民地的控制者
        parameters = {
            limit = {
                # 检测的条件
                # 这个行星拥有巨构且巨构的类型为星环基地
                num_pops >= 5
            }
        }
        mode = add
        mult = 1
        # 进行添加,添加的数量为1
    }
    max = 30
    # 最好写个最大值,添加的最大值为30,到达30后即使还有多的也不会再添加
    [[ZIJIANDE_JUGOU_FACTOR]
        mult = $ZIJIANDE_JUGOU_FACTOR$
    ]
    # 写一下这个,等下要用,最好为唯一值
}
