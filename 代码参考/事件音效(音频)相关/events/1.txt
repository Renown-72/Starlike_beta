# 事件音效,就类似于代码里sound用的东西
namespace = zijiande
country_event = {
    id = zijiande.1
    title = zijiande.1.title
    desc = zijiande.1.desc
    picture = GFX_evt_magic_huigui
    # 这里引用的图片是其它MOD的,你用这个演示代码不显示正确的图片的话那很正常
    is_triggered_only = yes
    #########################################
    show_sound = zijiande_Effects_1
    # 一般是用soundeffect的名字就行了,像这样
    #########################################
    option = {
        name = zijiande.1.a
    }
    option = {
        name = zijiande.1.b
        # ------------------------------ #
        play_sound = zijiande_Effects_1
        # 事件选项里也可以播放音频,用play_sound,你也可以拿事件做旮旯给木
        # 选了什么选项播放对应的音频()
        country_event = {
            id = zijiande.1
        }
        # ------------------------------ #
    }
    option = {
        name = zijiande.1.c
        # ------------------------------ #
        set_crisis_sound = zijiande_Effects_2
        # 还有一种是 将危机环境循环设置为当前效果
        # 这玩意应该是给类似于天灾的音效,这玩意在你不使用
        # stop_crisis_sound = yes,的时候,即使播放完毕退出游戏后载入也还会播放
        # 配合循环播放你可以在整什么东西的时候专门整个小曲玩
        country_event = {
            id = zijiande.1
        }
        # ------------------------------ #
    }
    option = {
        name = zijiande.1.d
        # ------------------------------ #
        stop_crisis_sound = yes
        # 停止播放set_crisis_sound的东西
        country_event = {
            id = zijiande.1
        }
        # ------------------------------ #
    }
}
# 就这样,快去玩吧~
