# 在agreement_term_values文件夹中编写我们的附庸条件和执行的效果
specialist_zijian = {
    term = specialist_type
    # term，使用原版即可表明你是一个特殊的附庸协议
    # 术语 = 专家类型
    triggered_loyalty_change = {
        # 满足条件附庸国的忠诚度变化，可以不写
        trigger = {
            # trigger里写条件，这里一般是写思潮的不同导致的忠诚度变化
            target = {
                # has_ethic代表拥有某种思潮，ethic_fanatic_materialist是极端唯物
                has_ethic = ethic_fanatic_materialist
            }
        }
        loyalty_change = 5
        # 忠诚度增加
        ai_acceptance = {
            # AI签订的意愿
            subject = 25
        }
    }
    activate_effect = {
        # 开启它的效果，这里是写等级了相关的了

        convert_to_specialist = specialist_zijian

        # 记得来到这里activate_effect，
        # 在convert_to_specialist里写入我们的specialist_subject_types值
        # 也就是specialist_zijian，这样就写好了

        # 使用convert_to_specialist来开启它的等级相关，
        # 这里的convert_to_specialist，需要填入specialist_zijian，但不是这个文件的specialist_zijian
        # 需要在specialist_subject_types另外编写，比如---我刚刚演示的，这里先不写
        # 因为我们还没有写specialist_subject_types里的开启效果，直接写他转换附庸协议会报错
    }
    # deactivate_effect = {
    #     # 停用效果，一般也和等级相关，一般是写停用后你拥有的什么BUFF消失了，这里可以不写
    #     # 一般只需要写activate_effect即可，它里面的convert_to_specialist开启等级相关
    # }
    #
    potential = {
        # potential，条件，一般这里需要再匹配一下附庸文件的值
        # 如preset_zijianfuyong
        agreement_preset = preset_zijianfuyong
    }
}
