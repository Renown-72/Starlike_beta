# 一个更简单的领袖制作方式
# 首先确认你得gfx文件夹下的models文件夹下portraits文件夹下有你得领袖图片
# 图片一般是.dds格式宽380PX，高可以自己测，这里使用的是494PX，或许可以使用PNG图片？
# 然后在gfx文件夹下的portraits文件夹下的portraits文件夹下创建一个txt文件注册你得领袖图片
# 使用portraits进行注册
portraits = {
    # zijian,这个是你领袖图的id，或者说key?反正很重要，需要唯一值
    # 使用textureFile引入你得图片,引入的路径正确鼠标指向它会出现图片
    zijian = { textureFile = gfx/models/portraits/NAME_MANTOUJIANONG.dds }
}
