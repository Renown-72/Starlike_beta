zijianlingxiu_2 = {
    # 先进行杀死领袖的判定
    # —————————————————————————————————————— #
    # 然后这里就需要通用的flag了
    every_owned_leader = {
        # 在开头检索我们的通用领袖的flag
        # 然后杀掉它
        limit = { has_leader_flag = zijianlingxiu_tongyong }
        kill_leader = {
            # show_notification = no的意思是杀死领袖的时候不进行通知
            show_notification = no
        }
    }
    # —————————————————————————————————————— #
    # 再进行创建领袖
    create_leader = {
        class = official
        # official是行政官，scientist是科研
        skill = 10
        name = zijianlingxiu_1
        custom_catch_phrase = zijianlingxiudejieshao
        immortal = yes
        gender = female
        species = owner_main_species
        set_age = 18
        event_leader = yes
        traits = {
            trait = leader_trait_zijian
        }
        effect = {
            # —————————————————————————————————————— #
            # 需要两个flag，首先是这个通用的flag
            # 因为我们要创建两个不同职能的同一个领袖
            # 这个通用的flag需要一样
            set_leader_flag = zijianlingxiu_tongyong
            # —————————————————————————————————————— #
            # 然后设置专属的flag-注意这个专属的flag不能和上面一样
            # 检测这个flag
            set_leader_flag = zijianlingxiu_keyanzhuanjia
            # —————————————————————————————————————— #
        }
    }
    # —————————————————————————————————————— #
    last_created_leader = {
        # 设置领袖的图片
        change_leader_portrait = zijian
    }
    # —————————————————————————————————————— #
}
# 这里说一下为什么要杀死领袖
# 因为我们要做一个可以通过通讯事件里的按钮切换领袖职能的效果
# 但我们已经召唤过的领袖不能和它同时出现啊,所以先进行一次杀死领袖的判定
# 然后再进行创建领袖，这样我们切换领袖的职能的时候
# 另一个职能的领袖就不会出现在我们的领袖池子里
# 这样创建领袖相关的就正好了
