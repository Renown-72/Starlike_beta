# 制作飞升
# 飞升在common文件夹下的ascension_perks文件夹中编写
ap_zijian = {
    potential = {
        is_ai = no
        # 这里条件也可以写限制某种起源什么的
        # has_origin = 后面的值写起源的名字
        NOT = {
            has_ascension_perk = ap_zijian
            # has_ascension_perk表示检测这个帝国是否有某种飞升，后面的值就写飞升的名字
            # 作用是防止这个飞升在选择后还可以选择的
            # 注意要写一个has_ascension_perk = ap_zijian
        }
    }
    possible = {}
    # 两个作用差不多，都是用来写条件的
    modifier = {
        # 可以在飞升里写选择后给与的BUFF修正
        country_naval_cap_mult = 1
    }
    on_enabled = {
        # on_enabled,这个飞升在启用后的效果，比如说执行一个事件
        country_event = { id = zijian.1 }
    }
    ai_weight = {
        # ai选择这个权重的几率，数值越大AI越喜欢选，点名排毒
        weight = 0
    }

    # 1
}
