# 编写一个联邦
# 联邦在common文件夹下的federation_types文件夹中编写
# 一般使用了federation_types和federation_perks和federation_laws和federation_law_categories文件
# federation_laws和federation_law_categories文件是联邦的部分按钮和界面
# 所以这里直接使用federation_types和federation_perks来编写一个联邦
default_federation_zijian = {
    base_intel = 60
    # 基础情报，直接复制一个即可，建议数字不要写太大，一般20到60即可
    icon = "GFX_default_federation"
    # 联邦的图片
    potential = {
        # 谁可以使用这个联邦？也就是哪个国家可以使用它
        # 一般都是always = yes，或者你也可以写个其他的限制
        # 这里的限制直接作用于国家
        always = yes
    }
    ownership_limits = default
    # 这里一般统一使用default，默认，这里是原版蠢驴写的舰船限制的，默认即可
    allow = {
        # 这里也是限制，优先级没potential高
        # 这里直接套用原版
        federation_check_for_subject_trigger = yes
        custom_tooltip = {
            fail_text = requires_actor_not_barbaric_despoilers
            NOT = { has_valid_civic = civic_barbaric_despoilers }
        }
    }
    levels = {
        # 联邦的等级，这里其实和附庸差不多
        # 注意这里原版复制过来是这样的
        level_1 = {
            # Starting rank
            # 但这里的experience值我建议你自己手写一个值
            experience = 1200 # Experience until next rank
            perks = {
                # 这里是联邦每升这一等级所获得的效果
                # 原版你直接写你会发现索引写不上去，其实无碍
                # 但你非要索引的话，可以去原版复制，这几个汉化放自己汉化文件夹里即可
                # 这样就能正确的索引了
                # 下面的我就不一个个写了，直接使用原版
                PACT_default_federation_zijian
            }
        }

        level_2 = {
            experience = 2400
            perks = {
                federation_fleet_size_1
                cohesion_join_1
                extra_envoy_1
            }
        }

        level_3 = {
            experience = 3600
            perks = {
                envoy_unity_1
                federation_fleet_speed_1
                extra_influence_1
            }
        }

        level_4 = {
            experience = 4800
            perks = {
                cohesion_failed_vote_1
                damage_crisis_1
                diplo_weight_1
            }
        }

        level_5 = {
            perks = {
                envoy_unity_2
                damage_crisis_2
                president_unity_1
            }
        }
    }
    on_create = {
        # on_create是联邦建立后联邦国家获得的效果和执行的事件，
        # 这里的效果其实就是加了标志FLAG,加了一个效果修正和设置基础的按钮
        # set_federation_law是设置这个联邦的基础按钮
        remove_federation_flag = enable_federation_cooldowns
        set_federation_law = centralization_minimal
        set_federation_law = succession_type_rotation
        set_federation_law = succession_term_years_20
        set_federation_law = fleet_contribution_none
        set_federation_law = federation_build_fleets_everyone
        set_federation_law = vote_weight_equal
        set_federation_law = declare_war_unanimous_vote
        set_federation_law = invite_members_unanimous_vote
        set_federation_law = kick_members_majority_vote
        set_federation_law = free_migration_no
        set_federation_law = treaties_separate_yes
        set_federation_law = allow_subjects_to_join_no
        set_federation_flag = enable_federation_cooldowns
        if = {
            limit = {
                NOT = {
                    has_modifier = diplo_fed_xpboost
                }
            }
            add_modifier = { modifier = diplo_fed_xpboost days = -1 mult = value:federation_xp_from_diplo_tradition }
        }
    }

    # 这里注意，虽然代码没有提示，
    # 但你还需要在gfx文件夹下的interface文件夹下的federation文件夹里设置你的联邦大图
    # 注册方式是把图片放到这个文件夹下面，写你的联邦名字 + _bg
    # default_federation_zijian + _bg = default_federation_zijian_bg
    # 这里格式是.dds的，原版使用的大小为 861 X 219 PX
    # 基本上这样就行了，进游戏看看
    # 这里删除ai_weight,我们不需要AI使用这个的权重

}
