namespace = zijiande
# 还是事件，这次将如何使用事件索引到我们所需要的目标身上，并为其执行效果
# 以这个MOD的领袖为例，
# 我们已知她拥有三个特殊的领袖特质，
# 领袖类别是指挥官，
# 并且拥有一个领袖FLAG，值为 ruler_yuanqianbei
# 那么我们如何使我们的事件找到她呢？
country_event = {
    id = zijiande.1
    hide_window = yes
    is_triggered_only = yes
    # 一般使用every，因为领袖是leader,那么直接输代码慢慢找即可
    # 一般使用every_owned_leader，
    # every，代表所有，leader代表领袖，owned代表控制的
    # 合起来就是所有控制着的领袖，但我们不能直接给所有领袖执行，这时候就需要给代码缩小索引的范围
    # 使用limit，注意，这里我们假设没有ruler_yuanqianbei这个领袖FLAG，那么我们需要使用前两个条件
    # 检查领袖的特质和领袖的类别，而领袖特质就是使用has_trait
    # has即为拥有，trait就是特质，合起来就是拥有的特质,这里直接填入我们这个领袖的特质
    # 然后锁定领袖的类别,由特质里的代码我们可以知道领袖的类别由leader_class控制，
    # 那我们可以使用leader_class来慢慢填入并找到我们所需要的那个条件
    # 因为是指挥官，所以使用commander
    # 如果我们找到的那个代码我们不确定它的作用怎么办？可以把鼠标移上去
    # 复制它并拿去翻译，翻译是检查领导者是否属于特定类别，
    # 你还拿不准的话还可以，复制这个代码，到源代码里全局搜索一下
    # 我们能直接看见源代码是如何使用的，那么就能确定我们使用的是正确的了
    # 那么我们给这个领袖加个灵能特质,一般使用add_trait，
    # 不知道可以去看插件的解释或者看源代码或者直接拿这个代码去翻译
    # 这里加个灵媒,leader_trait_psionic
    immediate = {
        every_owned_leader = {
            limit = {
                has_trait = leader_trait_mantoumonvhuashen
                has_trait = leader_trait_mantoushenglijian
                has_trait = leader_trait_mantoumeihuole
                leader_class = commander
            }
            add_trait = leader_trait_psionic
        }
    }
}
country_event = {
    id = zijiande.2
    hide_window = yes
    is_triggered_only = yes
    # 现在用事件2来写领袖
    # 首先我们需要锁定帝国,我们可以使用事件来寻找这个特殊的领袖在哪一个帝国身上,并给它添加一个标志FLAG
    # 我们需要在every_country的limit写入条件来找到这个领袖,那么可以使用any
    # any和every差不多,都是所有,但any一般用于trigger = {}和limit = {}里
    # 而every一般直接写在immediate里或者是option里
    # 这次我们假定这个领袖只有标志FLAG,没有其他知道的条件
    immediate = {
        every_country = {
            limit = {
                any_owned_leader = {
                    has_leader_flag = ruler_yuanqianbei
                }
            }
            set_country_flag = biaozhidiguo
        }
    }
}
country_event = {
    id = zijiande.3
    hide_window = yes
    is_triggered_only = yes
    # 然后用事件3来执行
    # 这里也是添加一个灵媒,leader_trait_psionic
    # 这里或许会有细心的人发现,我们的
    # every_owned_leader = {
    #     limit = { has_leader_flag = ruler_yuanqianbei }
    #     add_trait = leader_trait_psionic
    # }
    # 代码可以直接写在事件2的every_country下,是的,就是如此,这里事件2只是为了演示这么写罢了
    # 我们进游戏试试看
    immediate = {
        every_country = {
            limit = { has_country_flag = biaozhidiguo }
            every_owned_leader = {
                limit = { has_leader_flag = ruler_yuanqianbei }
                add_trait = leader_trait_psionic
            }
        }
    }
}
# 上面的三个事件已经完美的执行了,现在来说我们的事件4
country_event = {
    id = zijiande.4
    ######################################
    title = zijiande.2.title
    # 一般标题都是固定的
    desc = {
        # desc里多了trigger和text,text很明确,只是拿来写文本的
        # 而trigger我们可以看到它和我们使用过的trigger = {  }一模一样
        # 没错,这里也是填写条件的,在事件执行的时候,它会看你是否对应的条件,进而执行对应的文本
        # 然后条件不满足的文本就直接隐藏起来
        # 这里看文本可以知道,
        # 我们假设origin_default是地球联合国的专属起源,origin_lost_colony是人类联邦的专属起源
        # 那么当地球联合国执行这个事件的时候必定会出zijiande.2.a这个文本
        # 当地球联合国执行这个事件的时候必定会出zijiande.2.b这个文本
        trigger = { has_origin = origin_default }
        text = zijiande.2.a
    }
    desc = {
        trigger = { has_origin = origin_lost_colony }
        text = zijiande.2.b
    }
    ######################################
    is_triggered_only = yes
    ######################################
    option = {
        name = ok
    }
    option = {
        # 而option里的trigger也是一样,看你的条件来决定触发哪一些选项
        # 然后条件不满足的选项就直接隐藏起来
        # 那么当地球联合国执行这个事件的时候必定会出zijiande.2.title.a这个选项
        # 当地球联合国执行这个事件的时候必定会出zijiande.2.title.b这个选项
        # 让我们进游戏看看效果吧
        # 注：原版地球联合国使用的就是origin_default，而人类联邦使用的是origin_lost_colony
        trigger = { has_origin = origin_default }
        name = zijiande.2.title.a
    }
    option = {
        trigger = { has_origin = origin_lost_colony }
        name = zijiande.2.title.b
    }
    ######################################
}
