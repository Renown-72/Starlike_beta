namespace = chuangjian

planet_event = {
    # 首先是检测事件，使用行星事件planet_event进行检测
    id = chuangjian.1
    # 事件的id，使用事件名加点和数字组成 chuangjian + . + 数字，
    # 数字最好不要写太大，也不要由0开头
    is_triggered_only = yes
    # is_triggered_only由事件条件允许才执行，加了它以后就不会随随便便执行了
    hide_window = yes
    # hide_window关闭事件的弹窗，这里我们也不需要弹窗
    fire_only_once = yes
    # fire_only_once，只执行一次
    # 不然这位朋友，你也不想你得星系随随便便就执行吧
    trigger = {
        # trigger,事件执行所需的条件
        has_planet_flag = xingxi1_de_1_planet
        # 拥有这个行星xingxi1_de_1_planet，也就是星系1的行星
        # 它也代表了这个执行者使用的就是我们的系外星系的大地图行星进行的游戏
    }
    immediate = {
        # immediate立即执行
        root.owner = {
            # 执行效果，给控制着这个国家添加一个country_flag和一个global_event_target
            # 即scripted_effects脚本创建殖民地时使用的是它的全局global_event_target
            set_country_flag = chuangjiandekongzhi
            save_global_event_target_as = chuangjiandekongzhi
        }
    }
}
country_event = {
    # 给国家执行事件，使用country_event
    id = chuangjian.2
    # id，编写方式如上
    is_triggered_only = yes
    hide_window = yes
    fire_only_once = yes
    # 如上
    trigger = {
        # 这个是事件的条件
        # 这里也代表拥有chuangjiandekongzhi国家标志的那个国家，也就是
        # 使用系外恒星系进入游戏的那个国家
        has_country_flag = chuangjiandekongzhi
    }
    immediate = {
        # 执行效果
        capital_scope = {
            # 切换范围，将其从国家范围切换至行星和国家上，差不多应该是这个意思
            kaijuchuangjianxingxi = yes
            # 是否执行脚本，yes代表是，这个脚本就是我们编写的脚本名字
        }
    }
}
