namespace = zijiande
# 这里是一个写的差不多的事件
country_event = {
    id = zijiande.1
    hide_window = yes
    is_triggered_only = yes
    immediate = {
        # 这个事件先会找到一个不是AI的帝国,也就是玩家
        random_country = {
            limit = {
                is_ai = no
            }
            # 给你添加一个event_target
            save_event_target_as = origin_lost_colony_in_game
        }
        random_country = {
            limit = {
                # 假设这里的条件为人类联邦
                has_origin = origin_lost_colony
            }
            # 使用add_opinion_modifier来为他添加一个好感度
            add_opinion_modifier = {
                modifier = opinion_refused_human_2
                who = event_target:origin_lost_colony_in_game
            }
        }
    }
}
# 注意,我记得事件是无法为自己的帝国添加好感度的
# 自己的帝国好感度一般是使用trigger来进行判断是否获得的
