# 星界行动法令
# 需要在common/astral_actions中进行编写
# 事先声明,不管是你怎么写的星界行动法令
# 都需要进行一次探索裂隙才能正常的使用
# 与以前探索的无关,需要你这个星界行动法令出现后再探索一次(需要探索完毕)
action_zijiande = {
    sound = rift_open
    # 这个法令执行的声音,很浅显易懂,你可以用原版的音效或者是你自己注册的,注册音效下次再说

    # usages = 1
    # 不要写这玩意,这玩意是限制使用次数的,但你别写,等下我们用别的方法来限制使用次数

    unlock_threshold = 1
    # 你的科研船探索过几次星界裂隙才能让其解锁(需要探索完毕)

    cooldown = 5
    # 这个法令的冷却时间,演示我整快点,你自己写的时候看你自己

    activation_cost = {
        # 这个法令使用时候消耗的东西,比如这里是消耗的1合金,说一下如何整每次使用都能增加消耗
        # 直接复制原版的星界裂分代码
        cost = {
            alloys = 1
            multiplier = value:scaled_astral_action_cost|BASE|300|INCREASE|300|INCREASE_VARIABLE|action_zijiande_value|MAX|1200|
            # 这里的value:scaled_astral_action_cost是sv,我说过,但你不用管它,别动它,
            # BASE|300,基础是300,乘以1那就是第一次使用基础消耗是300合金,
            # INCREASE|300,增加300,再使用一次后就会增加300的消耗,比如第一次是300,那第二次就是600
            # BASE,INCREASE，INCREASE_VARIABLE,不要动它们
            # MAX|1200,最大消耗1200,很浅显易懂吧
            # num_rifts_splitted,等下给它改成我们设置的唯一值,这里不要使用原版的
            # 我们给它改成 action_zijiande_value，好了这样就行了
        }
    }

    allow = {
        is_ai = no

        # 注意你这里的值要和这里对应
        # 这里我们假设这个法令只能使用4次,这里你写<(小于)4就是限制4次数,
        # 就像这么写
        check_variable = {
            which = action_zijiande_value
            value < 4
        }

    }

    effect = {
        # 和上面的值要对应,这里的意思是每次使用都会更改
        # action_zijiande_value 的variable值,value是1的时候就加1,以此类推,是负1的时候就减1
        change_variable = {
            which = action_zijiande_value
            value = 1
        }

        # 这个星界行动法令执行后会获得一个300天的BUFF修正
        add_modifier = {
            modifier = spirit_of_ancestors_militarist_alloys_modifier_02
            days = 300
        }

    }

    picture = GFX_astral_action_astral_splitting
    # picture,这个法令的图片,需要你自己去interface里注册,已经有注册教程视频了,自己去看

    # ai_weight = {
    #     # AI的使用权重,你自己看着来写
    #     # 我上面写了is_ai = no禁止AI使用,所以这里就不写这个了
    #     weight = 100
    # }

    potential = {
        # 和allow一样也是限制条件,但建议不写这个,因为法令按钮要显示着,
        # 你条件写这里不满足条件就看不到你的法令
    }

}
